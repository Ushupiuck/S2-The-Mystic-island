; Multiplication:
	mulu.w	#6,d0
	; can be turned into
	add.w	d0,d0		; multiply by 6
	move.w	d0,d2
	add.w	d0,d0
	add.w	d2,d0
; ---------------------------------------------------------------------------
	mulu.w	#X,d0
	; can be turned into
; ---------------------------------------------------------------------------
	mulu.w	#X,d0
	; can be turned into
; ---------------------------------------------------------------------------
	mulu.w	#X,d0
	; can be turned into
; ---------------------------------------------------------------------------
	mulu.w	#$A,d0		; decimal 10
	; can be turned into
	move.w	d0,d1
	lsl.w	#3,d0
	add.w	d1,d0
	add.w	d1,d0
; ---------------------------------------------------------------------------
	mulu.w	#$2A,d1		; decimal 42
	; can be turned into
	moveq	#0,d2		; (optional) sanitize d2 (note: d0 was already in use, hence d2. The alternative was backing up & restoring d0 via SP, which would of been slower)
	move.w	d1,d2		; multiply by 42
	move.w	d1,d3
	lsl.w	#5,d1
	lsl.w	#3,d2
	add.w	d2,d1
	add.w	d3,d1
	add.w	d3,d1
; ---------------------------------------------------------------------------
	mulu.w	#$3C,d0		; decimal 60
 	; can be turned into
	add.w	d0,d0		; multiply by 60
	add.w	d0,d0
	move.w	d0,d1
	lsl.w	#4,d0
	sub.w	d1,d0
; ---------------------------------------------------------------------------
	mulu.w	#$80,d0		; decimal 128
	; can be turned into (exact math)
	andi.l	#$0000FFFF,d0
	lsl.l	#7,d0		; multiply by 128
	; or into (exact math, but only valid for values 0-$FF)
	lsl.w	#7,d0		; multiply by 128
	ext.l	d0
